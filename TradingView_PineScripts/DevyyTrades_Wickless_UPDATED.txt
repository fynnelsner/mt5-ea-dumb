getPipSize(goldMode, silverMode) =>
    bool isGold = str.contains(syminfo.ticker, "XAU") or str.contains(syminfo.ticker, "GOLD")
    bool isSilver = str.contains(syminfo.ticker, "XAG") or str.contains(syminfo.ticker, "SILVER")
    float goldVal = goldMode == "Standard ($0.10)" ? 0.1 : 0.01
    float silverVal = silverMode == "Standard ($0.01)" ? 0.01 : 0.001
    isGold ? goldVal : isSilver ? silverVal : (syminfo.mintick * (syminfo.type == "forex" or syminfo.type == "cfd" ? 10 : 1))

getTextSize(sizeStr) =>
    sizeStr == "Tiny" ? size.tiny : sizeStr == "Small" ? size.small : sizeStr == "Large" ? size.large : sizeStr == "Huge" ? size.huge : size.normal

// Calculates exact millisecond duration of a session for perfect center anchoring
getZoneDuration(sessStr) =>
    string zt = str.substring(sessStr, 0, 9)
    int sH = int(str.tonumber(str.substring(zt, 0, 2)))
    int sM = int(str.tonumber(str.substring(zt, 2, 4)))
    int eH = int(str.tonumber(str.substring(zt, 5, 7)))
    int eM = int(str.tonumber(str.substring(zt, 7, 9)))
    int startMinutes = sH * 60 + sM
    int endMinutes = eH * 60 + eM
    if endMinutes < startMinutes
        endMinutes += 24 * 60
    (endMinutes - startMinutes) * 60000

jsonPrice(v) =>
    na(v) ? "null" : str.tostring(v)

shouldSendWebhook(enableWebhooks, restrictTo15m) =>
    enableWebhooks and (not restrictTo15m or timeframe.period == "15")

buildStructureWebhook(direction, breakType, breakLevel, latestHigh, latestLow) =>
    '{"source":"wickless","event":"structure_update","symbol":"' + syminfo.ticker + '","timeframe":"' + timeframe.period + '","bar_time":' + str.tostring(time) + ',"data":{"direction":"' + direction + '","break_type":"' + breakType + '","break_level":' + jsonPrice(breakLevel) + ',"trend_after_break":"' + direction + '","latest_structure_high":' + jsonPrice(latestHigh) + ',"latest_structure_low":' + jsonPrice(latestLow) + ',"candle_open":' + jsonPrice(open) + ',"candle_high":' + jsonPrice(high) + ',"candle_low":' + jsonPrice(low) + ',"candle_close":' + jsonPrice(close) + '}}'

buildWicklessWebhook(candleType, entryAnchor, latestHigh, latestLow, trendDirection)
    jsonWickless = '{"source":"wickless","event":"wickless_signal","symbol":"' + syminfo.ticker + '","timeframe":"' + timeframe.period + '","bar_time":' + str.tostring(time) + ',"data":{"candle_type":"' + candleType + '","trend_direction":"' + trendDirection + '","entry_anchor":' + jsonPrice(entryAnchor) + ',"candle_open":' + jsonPrice(open) + ',"candle_high":' + jsonPrice(high) + ',"candle_low":' + jsonPrice(low) + ',"candle_close":' + jsonPrice(close) + ',"latest_structure_high":' + jsonPrice(latestHigh) + ',"latest_structure_low":' + jsonPrice(latestLow) + '}}'
    jsonWickless
